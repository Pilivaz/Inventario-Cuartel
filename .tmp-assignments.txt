function Assignments({assignments,movements,people,types,onReturn,onEdit,onResolvePending,onRetire,onLoss,onNew}:{assignments:Assignment[];movements:Movement[];people:Firefighter[];types:TypeRow[];onReturn:(a:Assignment)=>void;onEdit:(a:Assignment)=>void;onResolvePending:(a:Assignment)=>void;onRetire:(a:Assignment)=>void;onLoss:(a:Assignment)=>void;onNew:()=>void}) {
 const [filter,setFilter]=useState<'all'|'active'|'pending'|'closed'>('all');
 const list=assignments.filter(a=>filter==='all'||(filter==='active'&&a.status==='active')||(filter==='pending'&&a.status==='pending_review')||(filter==='closed'&&a.status!=='active'&&a.status!=='pending_review'));
 const statusLabel=(status:string)=>status==='lost'?'Perdida':status==='returned'?'Devuelta':status==='pending_review'?'Pendiente de revisión':status==='retired'?'Dada de baja':'Activa';
 return <><PageHeading eyebrow="TRAZABILIDAD" title="Asignaciones" desc="Entregas, devoluciones e incidentes del equipamiento." action={<button className="primary" onClick={onNew}><Plus size={17}/> Nueva asignación</button>}/><div className="toolbar"><div className="filter-tabs">{[['all','Todas'],['active','Activas'],['pending','Pendientes'],['closed','Cerradas']].map(([v,l])=><button className={filter===v?'selected':''} key={v} onClick={()=>setFilter(v as typeof filter)}>{l}</button>)}</div><span className="results-count">{list.length} asignaciones</span></div><div className="table-wrap"><table><thead><tr><th>BOMBERO</th><th>ELEMENTO</th><th>FECHA DE ENTREGA</th><th>ESTADO</th><th>FECHA DE CIERRE</th><th>ACCIÓN</th></tr></thead><tbody>{list.map(a=><tr key={a.id}><td><b>{a.firefighters?.full_name||'—'}</b><small className="td-sub">Legajo {a.firefighters?.personnel_no||'—'}</small></td><td>{a.equipment_units?.equipment_types?.name||a.equipment_units?.unit_code||'—'}<small className="td-sub">{a.equipment_units?.unit_code||''}</small></td><td>{fmtDate(a.delivered_at)}</td><td><span className={`pill ${a.status==='active'?'pill-red':a.status==='pending_review'?'pill-amber':'pill-muted'}`}>{statusLabel(a.status)}</span></td><td>{fmtDate(a.returned_at||undefined)}</td><td>{a.status==='active'?<div className="line-actions"><button className="secondary small-button" onClick={()=>onEdit(a)}><Pencil size={13}/> Editar</button><button className="text-button small-button" onClick={()=>onReturn(a)}>El bombero ya no lo tiene</button><button className="danger-link" onClick={()=>onLoss(a)}>Perdido</button><button className="danger-link" onClick={()=>onRetire(a)}>Dar de baja</button></div>:a.status==='pending_review'?<button className="secondary small-button" onClick={()=>onResolvePending(a)}>Resolver revisión</button>:null}</td></tr>)}</tbody></table>{!list.length&&<Empty title="Sin asignaciones" text="Al registrar una entrega, aparecerá en este historial."/>}</div><section className="panel movement-audit"><div className="panel-head"><div><h2>Historial de movimientos</h2><p>Auditoría registrada en Supabase</p></div><span className="muted-tag">{movements.length} registros</span></div>{movements.length?<div className="movement-list">{movements.slice(0,50).map(m=><article className="movement-row" key={m.id}><div><b>{movementLabel(m.movement_type)}</b><span>{m.firefighters?.full_name||'Cuartel'}{m.firefighters?.personnel_no?` · Legajo ${m.firefighters.personnel_no}`:''} · {m.equipment_units?.equipment_types?.name||m.equipment_units?.unit_code||String(m.details?.template_name||m.details?.after_name||'')}</span></div><time>{fmtDate(m.created_at)}</time>{m.notes&&<small>{m.notes}</small>}</article>)}</div>:<Empty title="Sin movimientos registrados" text="Las operaciones auditadas aparecerán aquí."/>}</section><p className="footnote">La tabla conserva todo el historial; esta página muestra las últimas 50 entradas de auditoría.</p><span className="sr-only">{people.length} personas y {types.length} tipos disponibles</span></>;
}
