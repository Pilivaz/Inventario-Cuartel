function LegacyProfileLayout({rows,reviews={},reviewSchemaReady=false,active=[],onReturn,onEditAssignment,onResolvePending,onRetireAssignment,onEditLegacy,onLegacyCustody,onLoss}:{rows:Array<Pick<LegacyRow,'id'|'source_sheet'|'source_row'|'raw_data'>>;reviews?:Record<string,LegacyRowReview>;reviewSchemaReady?:boolean;active?:Assignment[];onReturn?:(a:Assignment)=>void;onEditAssignment?:(a:Assignment)=>void;onResolvePending?:(a:Assignment)=>void;onRetireAssignment?:(a:Assignment)=>void;onEditLegacy?:(row:LegacyRow)=>void;onLegacyCustody?:(row:LegacyRow)=>void;onLoss?:(a:Assignment)=>void}) {
 const sectionOf=(row:typeof rows[number])=>row.raw_data._section||row.source_sheet.replace('Ficha individual · ','');

 const elements=sorted.filter(row=>sectionOf(row)!=='Traje de gala'&&sectionOf(row)!=='Traje de fajina');

 </div>;
